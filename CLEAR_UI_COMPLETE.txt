================================================================================
                            QUICK START GUIDE
================================================================================

1. INITIALIZE:
   auto& ui = SZM::UI::ClearUIManager::GetInstance();
   ui.Initialize();

2. ADD SECTION:
   auto section = std::make_shared<UISection>("Title", "🎯");
   section->SetContent([]() {
       ImGui::Text("Your UI code here");
   });
   ui.AddSectionToTab(TabType::Dashboard, section);

3. RENDER:
   ui.RenderUI();

================================================================================
                            INTEGRATION STEPS
================================================================================

1. Include header in main_desktop.cpp:
   #include "UI/ClearUIManager.hpp"

2. Initialize in Application::Init():
   auto& ui = SZM::UI::ClearUIManager::GetInstance();
   ui.Initialize();

3. Call RenderUI() in Application::TickSystems():
   ui.RenderUI();

4. Add sections to tabs as needed:
   auto section = std::make_shared<UISection>("Title");
   section->SetContent([]() { /* UI code */ });
   ui.AddSectionToTab(TabType::Dashboard, section);

5. Rebuild:
   cmake --build build-full --parallel
