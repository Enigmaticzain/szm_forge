cmake_minimum_required(VERSION 3.21)
project(SZM_Forge_AutoBuild VERSION 0.2.1 LANGUAGES CXX C)

# ============================================================
#  Auto-Discovery Build System
#  Automatically detects and includes new modules
# ============================================================

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# ============================================================
#  Build Configuration
# ============================================================
set(SZM_BUILD_FULL_DESKTOP ON CACHE BOOL "Always build full desktop" FORCE)
set(SZM_AUTO_DISCOVER_MODULES ON CACHE BOOL "Auto-discover modules" FORCE)

# ============================================================
#  Platform Definitions
# ============================================================
if(WIN32)
    add_compile_definitions(SZM_PLATFORM_WINDOWS)
elseif(UNIX AND NOT APPLE)
    add_compile_definitions(SZM_PLATFORM_LINUX)
elseif(APPLE)
    add_compile_definitions(SZM_PLATFORM_MACOS)
endif()

# ============================================================
#  Compiler Warnings
# ============================================================
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    add_compile_options(-Wall -Wextra -Wpedantic)
elseif(CMAKE_CXX_COMPILER_ID MATCHES "MSVC")
    add_compile_options(/W4)
    add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
endif()

# ============================================================
#  Auto-Discovery Function
# ============================================================
function(discover_modules)
    set(SRC_DIR "${CMAKE_SOURCE_DIR}/src")
    set(DISCOVERED_MODULES "")
    set(MODULE_TARGETS "")
    
    # Scan src/ for subdirectories with CMakeLists.txt
    file(GLOB SUBDIRS RELATIVE "${SRC_DIR}" "${SRC_DIR}/*")
    
    foreach(SUBDIR ${SUBDIRS})
        set(MODULE_PATH "${SRC_DIR}/${SUBDIR}")
        
        if(IS_DIRECTORY "${MODULE_PATH}" AND EXISTS "${MODULE_PATH}/CMakeLists.txt")
            # Skip hidden directories and known non-modules
            if(NOT SUBDIR MATCHES "^\\..*")
                list(APPEND DISCOVERED_MODULES "${SUBDIR}")
                message(STATUS "[AUTO-DISCOVER] Found module: ${SUBDIR}")
            endif()
        endif()
    endforeach()
    
    set(DISCOVERED_MODULES "${DISCOVERED_MODULES}" PARENT_SCOPE)
endfunction()

# ============================================================
#  External Dependencies
# ============================================================
find_package(nlohmann_json QUIET)
if(NOT nlohmann_json_FOUND)
    include(FetchContent)
    FetchContent_Declare(

    )
    FetchContent_MakeAvailable(nlohmann_json)
endif()

# GLFW
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(${CMAKE_SOURCE_DIR}/external/glfw EXCLUDE_FROM_ALL)

# Eigen3
find_package(Eigen3 3.4 REQUIRED NO_MODULE)

# OpenGL
find_package(OpenGL REQUIRED)

# GLAD
set(GLAD_DIR ${CMAKE_SOURCE_DIR}/external/glad)
add_library(glad STATIC ${GLAD_DIR}/src/gl.c)
target_include_directories(glad PUBLIC ${GLAD_DIR}/include)

# ImGui
set(IMGUI_DIR ${CMAKE_SOURCE_DIR}/external/imgui)
add_library(imgui STATIC
    ${IMGUI_DIR}/imgui.cpp
    ${IMGUI_DIR}/imgui_draw.cpp
    ${IMGUI_DIR}/imgui_tables.cpp
    ${IMGUI_DIR}/imgui_widgets.cpp
    ${IMGUI_DIR}/backends/imgui_impl_glfw.cpp
    ${IMGUI_DIR}/backends/imgui_impl_opengl3.cpp
)
target_include_directories(imgui PUBLIC
    ${IMGUI_DIR}
    ${IMGUI_DIR}/backends
)
target_compile_definitions(imgui PUBLIC IMGUI_IMPL_OPENGL_LOADER_GLAD)
target_link_libraries(imgui PUBLIC glfw glad OpenGL::GL)

# ============================================================
#  Auto-Discover and Add Modules
# ============================================================
discover_modules()

# Define module discovery order (dependencies first)
set(MODULE_ORDER
    Simulation
    Data
    Input
    Materials
    Math
    Geometry
    Physics
    Assembly
    FEA
    Thermal
    Electrical
    Graphics
    UI
    Core
    Workshop3D
    WebViewer
    CAD
    AI
    WebAPI
)

# Add modules in dependency order
foreach(MODULE ${MODULE_ORDER})
    set(MODULE_PATH "${CMAKE_SOURCE_DIR}/src/${MODULE}")
    if(EXISTS "${MODULE_PATH}/CMakeLists.txt")
        message(STATUS "[BUILD] Adding module: ${MODULE}")
        add_subdirectory("${MODULE_PATH}")
    endif()
endforeach()

# Add any remaining discovered modules not in the order list
foreach(MODULE ${DISCOVERED_MODULES})
    list(FIND MODULE_ORDER "${MODULE}" FOUND_INDEX)
    if(FOUND_INDEX EQUAL -1)
        set(MODULE_PATH "${CMAKE_SOURCE_DIR}/src/${MODULE}")
        if(EXISTS "${MODULE_PATH}/CMakeLists.txt")
            message(STATUS "[BUILD] Adding discovered module: ${MODULE}")
            add_subdirectory("${MODULE_PATH}")
        endif()
    endif()
endforeach()

#  Collect All Module Targets
# ============================================================

    SZM_Geometry_Module
